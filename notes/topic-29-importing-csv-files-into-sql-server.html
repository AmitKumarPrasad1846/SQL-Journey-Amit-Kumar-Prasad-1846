Importing CSV Files into SQL Server

Load bulk delimited text data into SQL Server tables using BULK INSERT and SSMS wizards.

Open in Playground Download Script

Concept Overview

Loading flat files like CSV or TSV into relational tables is a common ETL requirement. The T-SQL `BULK INSERT` command streams external data directly into a table, specifying row terminators (`\n`), field terminators (`,`), and header skip lines (`FIRSTROW = 2`).

Hands-on SQL Script

29 SQL Query Importing CSV files.sql 
--Import CSV

--dbo means database object.
use Google
--CREATING TABLE RABBIT
create table Rabbit
(id int, lastname varchar(255), firstname varchar(255), middlename varchar(255),
suffix varchar(255))
--METHOD 1 -- WITHOUT ANY LIMIT  _ COMPLETE IMPORT

--giving only firstrow.
bulk insert dbo.rabbit
from 'C:\sql file\Tiger.csv'
with
(
format = 'CSV',
firstrow = 2
)

select * from rabbit

--METHOD 2 -- WITH LIMIT  _ SELECTED IMPORT
--CREATING ANOTHER TABLE
create table parrot
(Id int, lastname varchar(255), firstname varchar(255), middlename varchar(255),
suffix varchar(255))

select * from parrot


--giving first and last row.
bulk insert dbo.parrot
from 'C:\sql file\Tiger.csv'
with
(
format = 'CSV',
firstrow = 2,
lastrow = 21
)

select * from parrot;
